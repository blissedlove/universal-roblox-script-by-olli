local function Slider(name, min, max, default, callback)
    Label(name .. ": " .. default)

    local value = default
    local button = Button("Decrease", function()
        value = math.max(min, value - 1)
        callback(value)
    end)

    local increase = Button("Increase", function()
        value = math.min(max, value + 1)
        callback(value)
    end)

    return function()
        return value
    end
end

local function Toggle(name, initial, callback)
    local state = initial
    local button

    button = Button(name .. ": " .. tostring(state), function()
        state = not state
        button.Text = name .. ": " .. tostring(state)
        callback(state)
    end)

    return button
end

-- Combat: target selection for an authorized practice system.
local function GetTarget()
    local closest, shortest = nil, Config.FOV

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local character = player.Character
            local humanoid = character
                and character:FindFirstChildOfClass("Humanoid")
            local part = character
                and character:FindFirstChild(Config.AimPart)

            if humanoid and humanoid.Health > 0 and part then
                local position, visible =
                    Camera:WorldToViewportPoint(part.Position)

                if visible then
                    local center = Camera.ViewportSize / 2
                    local distance = (
                        Vector2.new(position.X, position.Y) - center
                    ).Magnitude

                    if distance < shortest then
                        closest = part
                        shortest = distance
                    end
                end
            end
        end
    end

    return closest
end

-- Visuals: highlights for your own experience.
local Highlights = {}

local function UpdateHighlights()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local character = player.Character
            local highlight = Highlights[player]

            if Config.BoxESP then
                if not highlight or highlight.Parent ~= character then
                    if highlight then highlight:Destroy() end

                    highlight = Instance.new("Highlight")
                    highlight.FillTransparency = 0.85
                    highlight.OutlineColor =
                        Color3.fromRGB(0, 190, 255)
                    highlight.Parent = character
                    Highlights[player] = highlight
                end
            elseif highlight then
                highlight:Destroy()
                Highlights[player] = nil
            end
        end
    end
end

end

local function ConfigTab()
    ClearContent()
    Label("CONFIGURATION")

    Button("Save Config", function()
        -- Save current settings in memory.
        GUI:SetAttribute("AimEnabled", Config.AimEnabled)
        GUI:SetAttribute("AimPart", Config.AimPart)
        GUI:SetAttribute("Smoothness", Config.Smoothness)
        GUI:SetAttribute("FOV", Config.FOV)
        GUI:SetAttribute("BoxESP", Config.BoxESP)
        GUI:SetAttribute("Skeleton", Config.Skeleton)
        GUI:SetAttribute("Tracers", Config.Tracers)
    end)

    Button("Load Config", function()
        for key in pairs(Config) do
            local value = GUI:GetAttribute(key)
            if value ~= nil then
                Config[key] = value
            end
        end
    end)

    Label("Settings persist during this session.")
end

local function MiscTab()
    ClearContent()
    Label("MISCELLANEOUS")

    Button("Unload Script", function()
        Unloaded = true

        for _, connection in ipairs(Connections) do
            connection:Disconnect()
        end

        for _, highlight in pairs(Highlights) do
            highlight:Destroy()
        end

        GUI:Destroy()
    end)
end
