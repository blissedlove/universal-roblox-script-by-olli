    table.insert(Connections, connection)
    return connection
end

        end
    end)

-- ESP
local function RemoveESP(target)
    if ESPObjects[target] then
        for _, object in ipairs(ESPObjects[target]) do
            object:Destroy()
        end
        ESPObjects[target] = nil
    end
end

local function AddESP(target)
    if target == Player or not target.Character then return end

    RemoveESP(target)

    local character = target.Character
    local root = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")

    if not root or not humanoid then return end

    local highlight = Instance.new("Highlight")
    highlight.Adornee = character
    highlight.FillColor = Color3.fromRGB(140, 70, 255)
    highlight.OutlineColor = Color3.new(1, 1, 1)
    highlight.FillTransparency = 0.65
    highlight.Parent = character

    local billboard = Instance.new("BillboardGui")
    billboard.Adornee = root
    billboard.Size = UDim2.fromOffset(190, 40)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Parent = GUI

    local label = Instance.new("TextLabel")
    label.Size = UDim2.fromScale(1, 1)
    label.BackgroundTransparency = 1
    label.TextColor3 = Color3.new(1, 1, 1)
    label.TextStrokeTransparency = 0
    label.Font = Enum.Font.GothamBold
    label.TextSize = 12
    label.Parent = billboard

    ESPObjects[target] = {highlight, billboard, label}
end

local function RefreshESP()
    for _, target in ipairs(Players:GetPlayers()) do
        if target ~= Player then
            if Settings.ESP then
                AddESP(target)
            else
                RemoveESP(target)
            end
        end
    end
end

local unloadButton
unloadButton = Button("UNLOAD TOOLKIT", 400, function()
    if Unloaded then return end
    Unloaded = true

    -- Restore movement
    local character = Player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")

    if humanoid then
        humanoid.WalkSpeed = OriginalWalkSpeed
        humanoid.JumpPower = OriginalJumpPower
    end

    -- Disconnect every saved connection
    for _, connection in ipairs(Connections) do
        if connection.Connected then
            connection:Disconnect()
        end
    end

    -- Remove ESP
    for target in pairs(ESPObjects) do
        RemoveESP(target)
    end

    GUI:Destroy()
end, Color3.fromRGB(155, 45, 55))

-- TARGET PART
local function GetAimPart(character)
    if Settings.AimPart == "Head" then
        return character:FindFirstChild("Head")
    end

    return character:FindFirstChild("UpperTorso")
        or character:FindFirstChild("Torso")
        or character:FindFirstChild("HumanoidRootPart")
end

-- TARGET SELECTION
local function GetClosestTarget()
    local mousePosition = UIS:GetMouseLocation()
    local closestPart = nil
    local shortestDistance = Settings.FOV

    for _, target in ipairs(Players:GetPlayers()) do
        if target ~= Player and target.Character then
            local character = target.Character
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            local part = GetAimPart(character)

            local sameTeam = Player.Team ~= nil
                and target.Team == Player.Team

            if humanoid and humanoid.Health > 0 and part and not sameTeam then
                local screen, visible =
                    Camera:WorldToViewportPoint(part.Position)

                if visible then
                    local distance = (
                        Vector2.new(screen.X, screen.Y) - mousePosition
                    ).Magnitude

                    if distance < shortestDistance then
                        shortestDistance = distance
                        closestPart = part
                    end
                end
            end
        end
    end

    return closestPart
end

-- INPUT
Connect(UIS.InputBegan, function(input)
    if input.KeyCode == Enum.KeyCode.RightControl then
        Main.Visible = not Main.Visible
    end

    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        AimHeld = true
    end
end)

Connect(UIS.InputEnded, function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        AimHeld = false
    end
end)

-- PLAYER CONNECTIONS
Connect(Players.PlayerAdded, function(target)
    Connect(target.CharacterAdded, function()
        task.wait(1)
        if not Unloaded and Settings.ESP then
            AddESP(target)
        end
    end)
end)

Connect(Players.PlayerRemoving, RemoveESP)

for _, target in ipairs(Players:GetPlayers()) do
    if target ~= Player then
        Connect(target.CharacterAdded, function()
            task.wait(1)
            if not Unloaded and Settings.ESP then
                AddESP(target)
            end
        end)
    end
end

-- MAIN LOOP
Connect(RunService.RenderStepped, function()
    if Unloaded then return end

    Camera = workspace.CurrentCamera

    local mousePosition = UIS:GetMouseLocation()
    Circle.Position = UDim2.fromOffset(mousePosition.X, mousePosition.Y)
    Circle.Visible = Settings.Aim

    -- Update ESP text
    for target, objects in pairs(ESPObjects) do
        local character = target.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")

        if root and humanoid then
            local distance = (root.Position - Camera.CFrame.Position).Magnitude

            objects[3].Text = string.format(
                "%s | HP: %d | %.0f studs",
                target.DisplayName,
                humanoid.Health,
                distance
            )
        end
    end

    -- Aim assist for your own training experience
    if Settings.Aim and AimHeld then
        local target = GetClosestTarget()

        if target then
            local desired = CFrame.lookAt(
                Camera.CFrame.Position,
                target.Position
            )

            local alpha = math.clamp(
                1 - math.exp(-Settings.Smoothness * 0.1),
                0,
                1
            )

            Camera.CFrame = Camera.CFrame:Lerp(desired, alpha)
        end
    end

    -- Movement
    local character = Player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")

    if humanoid then
        humanoid.WalkSpeed = Settings.Sprint and Settings.Speed or 16
        humanoid.UseJumpPower = true
        humanoid.JumpPower = Settings.Jump
    end
end)
